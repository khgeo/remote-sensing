# មេរៀនទី១៣៖ ការវាយតម្លៃភាពត្រឹមត្រូវ

!!! info "ព័ត៌មានមេរៀន"
    **រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង** · មេរៀនទី ១៣ ក្នុងចំណោម ១៥

ផែនទីគម្របដីមួយ មិនថាបានមកពីការចាត់ថ្នាក់គ្មានការណែនាំ ([មេរៀនទី១១](lesson-11.md)) ឬមានការណែនាំ ([មេរៀនទី១២](lesson-12.md)) សុទ្ធតែជា **ការទាយ** មិនមែនការវាស់ដោយផ្ទាល់ទេ។ សំណួរដែលអ្នកសម្រេចចិត្តគ្រប់រូបតែងសួរគឺ៖ «ខ្ញុំគួរជឿទុកចិត្តលើផែនទីនេះប៉ុនណា?»

មេរៀននេះជាចម្លើយជាប្រព័ន្ធចំពោះសំណួរនោះ។ យើងនឹងសិក្សា **តារាងច្របូកច្របល់ (Confusion Matrix)** ដែលជាឧបករណ៍ស្នូលសម្រាប់រាយការណ៍ភាពត្រឹមត្រូវនៃការចាត់ថ្នាក់រូបភាពគ្រប់ការសិក្សា និងអត្ថន័យលេខនីមួយៗដែលវាបង្ហាញ។

## ១៣.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. បង្កើត និងអានតារាងច្របូកច្របល់។
2. គណនា Overall Accuracy Producer's Accuracy និង User's Accuracy។
3. ពន្យល់ថាហេតុអ្វី Kappa coefficient ចាំបាច់ ក្រៅពីភាពត្រឹមត្រូវសរុប។
4. ជ្រើសយុទ្ធសាស្ត្រយកគំរូផ្ទៀងផ្ទាត់ត្រឹមត្រូវ (ចៃដន្យ ស្រទាប់ ជាដើម)។
5. កំណត់ភាពខុសគ្នារវាងកំហុសនៃការធ្វើផែនទី និងកំហុសនៃការវាស់វែងភាពត្រឹមត្រូវខ្លួនឯង។

## ១៣.២. ស្ថានភាពបើកមេរៀន៖ ផែនទីមួយណាគួរជឿជាក់ជាង?

អ្នកស្រាវជ្រាវពីរនាក់ធ្វើផែនទីព្រៃឈើលើតំបន់ដូចគ្នា។ ម្នាក់ប្រើ K-means ម្នាក់ទៀតប្រើចម្ងាយអប្បបរមាដែលមានគំរូបណ្ដុះបណ្ដាល។ ផែនទីទាំងពីរមើលទៅសមហេតុផល ប៉ុន្តែខុសគ្នាបន្តិចត្រង់ព្រំតំបន់។

ដោយគ្មានការវាយតម្លៃជាផ្លូវការ គ្មាននរណាអាចនិយាយថាមួយណាល្អជាងបានទេ។ ដំណោះស្រាយគឺ **យកគំរូផ្ទៀងផ្ទាត់ឯករាជ្យ** (ដែលមិនប្រើក្នុងការបណ្ដុះបណ្ដាល ឬការគណនា K-means) រួចប្រៀបធៀបលទ្ធផលចាត់ថ្នាក់ជាមួយក្រប់ដីពិតនៅចំណុចទាំងនោះ។ លទ្ធផលនៃការប្រៀបធៀបនោះគឺ **តារាងច្របូកច្របល់**។

សំណួរដែលមេរៀននេះឆ្លើយ៖ តារាងច្របូកច្របល់សរសេរយ៉ាងណា? តើលេខអ្វីខ្លះដែលយើងគណនាពីវា ហើយអត្ថន័យរបស់វាជាអ្វី? ហើយហេតុអ្វីភាពត្រឹមត្រូវសរុបតែម្នាក់ឯង អាចបំភាន់?

## ១៣.៣. ទ្រឹស្ដីស្នូល

### ១៣.៣.១. តារាងច្របូកច្របល់

**តារាងច្របូកច្របល់ (Confusion Matrix)** ប្រៀបធៀប **ក្រប់ដីព្យាករ** (ពីការចាត់ថ្នាក់) ជាមួយ **ក្រប់ដីពិត** (ពីចំណុចផ្ទៀងផ្ទាត់ទីវាល ឬការបកស្រាយគុណភាពខ្ពស់ជាង)៖

<figure markdown>
--8<-- "assets/svg/r13-confusion.svg"
<figcaption>រូបទី១៣.១៖ កាយវិភាគនៃតារាងច្របូកច្របល់ (Confusion matrix)។ អង្កត់ទ្រូងតំណាងចំនួនក្រឡាដែលចាត់ថ្នាក់ត្រូវ ខណៈក្រៅអង្កត់ទ្រូងបង្ហាញការច្រឡំរវាងថ្នាក់ណាមួយទៅថ្នាក់ណាមួយ។</figcaption>
</figure>

**ការអាន**៖ ជួរដេកតំណាងក្រប់ដីពិត ជួរឈរតំណាងក្រប់ដីព្យាករ (ឬផ្ទុយវិញ អាស្រ័យលើអ្នកនិពន្ធ ត្រូវពិនិត្យកំណត់សម្គាល់ជានិច្ច)។ **អង្កត់ទ្រូង** បង្ហាញចំនួនចំណុចដែលចាត់ថ្នាក់ត្រូវ (ក្រប់ដីព្យាករ = ក្រប់ដីពិត)។ **ក្រៅអង្កត់ទ្រូង** បង្ហាញការច្រឡំ៖ ជួរដេក «ព្រៃឈើ» ជួរឈរ «ស្រែ» មានតម្លៃ ២ មានន័យថា ចំណុចពិត ២ ដែលជាព្រៃឈើ ត្រូវបានចាត់ថ្នាក់ខុសទៅជាស្រែ។

!!! example "ពិសោធន៍៖ បង្កើតតារាងច្របូកច្របល់"
    ប្ដូរចំនួនគំរូ ហើយចុចប៊ូតុងដើម្បីបង្កើតតារាងថ្មី។ សង្កេតកន្លែងណាដែលមានការច្រឡំ (ក្រឡាក្រៅអង្កត់ទ្រូងដែលមានតម្លៃ)។

<div class="sim" data-sim="rs-confmat"></div>

### ១៣.៣.២. ភាពត្រឹមត្រូវសរុប

**Overall Accuracy (OA)** ជាតួលេខសង្ខេបសាមញ្ញបំផុត៖

> **OA = (ផលបូកអង្កត់ទ្រូង) ÷ (ផលបូកសរុបទាំងអស់) × ១០០%**

វាឆ្លើយសំណួរ «ជារួម តើចំណុចប៉ុន្មានភាគរយត្រូវបានចាត់ថ្នាក់ត្រូវ?» ប៉ុន្តែវាបាំង **ភាពខុសគ្នារវាងថ្នាក់**។ ថ្នាក់មួយអាចត្រឹមត្រូវ ១០០% ខណៈថ្នាក់មួយទៀតត្រឹមត្រូវត្រឹមតែ ៦០% ហើយ OA នៅតែអាចមើលទៅល្អ បើថ្នាក់ដែលត្រឹមត្រូវខ្ពស់មានចំនួនច្រើនជាង។ នេះជាមូលហេតុដែលយើងត្រូវការតួលេខតាមថ្នាក់ផងដែរ។

### ១៣.៣.៣. ភាពត្រឹមត្រូវតាមថ្នាក់៖ Producer's និង User's

ពីតារាងច្របូកច្របល់ គណនាបានពីរតួលេខសំខាន់ក្នុងមួយថ្នាក់៖

- **Producer's Accuracy (PA)**៖ គណនាពីជួរដេក (ក្រប់ដីពិត)។ ចំពោះចំណុចដែលជាព្រៃឈើពិត ប៉ុន្មានភាគរយត្រូវបានចាត់ថ្នាក់ត្រូវជាព្រៃឈើ? វាវាស់ **ការខកខាន (omission error)**៖ ១០០% − PA = ភាគរយនៃព្រៃឈើពិតដែលខកខានមិនបានចាត់ថ្នាក់ត្រូវ។
- **User's Accuracy (UA)**៖ គណនាពីជួរឈរ (ក្រប់ដីព្យាករ)។ ចំពោះចំណុចដែលផែនទីនិយាយថាជាព្រៃឈើ ប៉ុន្មានភាគរយជាព្រៃឈើពិតប្រាកដ? វាវាស់ **ការបន្ថែមខុស (commission error)**៖ ១០០% − UA = ភាគរយនៃចំណុចដែលចាត់ខុសចូលមកក្នុងថ្នាក់នេះ។

ភាពខុសគ្នារវាងទាំងពីរសំខាន់ណាស់សម្រាប់ការសម្រេចចិត្ត៖ បើអ្នកគ្រោងប្រើផែនទីដើម្បី **រកព្រៃឈើទាំងអស់** (ឧ. សម្រាប់ការការពារ) PA សំខាន់ជាង។ បើអ្នកគ្រោង **ជឿជាក់លើផ្ទៃដីដែលសម្គាល់ថាជាព្រៃឈើ** (ឧ. សម្រាប់ការវិនិយោគ) UA សំខាន់ជាង។

### ១៣.៣.៤. Kappa Coefficient

ភាពត្រឹមត្រូវសរុបមានបញ្ហាមួយទៀត៖ វាមិនគិតគូរថា **ការទាយចៃដន្យក៏អាចផ្ដល់ភាពត្រឹមត្រូវខ្ពស់ដែរ** បើថ្នាក់មួយគ្របដណ្ដប់ភាគច្រើននៃទិន្នន័យ។ ឧទាហរណ៍ បើ ៨០% នៃតំបន់សិក្សាជាព្រៃឈើ ការទាយថា «ព្រៃឈើ» សម្រាប់គ្រប់ចំណុច នឹងផ្ដល់ភាពត្រឹមត្រូវសរុប ៨០% ដោយគ្មានការវិភាគអ្វីសោះ។

!!! example "ពិសោធន៍៖ ហេតុអ្វីត្រូវការ Kappa"
    ប្ដូរ «ថ្នាក់ភាគច្រើន» ឲ្យខ្ពស់ (ដូចព្រៃឈើគ្របដណ្ដប់ភាគច្រើន) ហើយសង្កេតថា Kappa ទាបជាង OA ខ្លាំងប៉ុនណា។

<div class="sim" data-sim="rs-kappa"></div>

**Kappa Coefficient (κ)** កែតម្រូវសម្រាប់ការឯកភាពដោយចៃដន្យ៖

> **κ = (OA − pe) ÷ (1 − pe)**

ដែល `pe` ជាភាពត្រឹមត្រូវដែលរំពឹងទុកពីការទាយចៃដន្យ គិតពីសមាមាត្រនៃថ្នាក់នីមួយៗក្នុងតារាង។ ការបកស្រាយប្រហាក់ប្រហែល (Landis & Koch)៖

| Kappa | ការបកស្រាយ |
|---|---|
| < ០ | អាក្រក់ជាងចៃដន្យ |
| ០ ដល់ ០,២ | ខ្សោយណាស់ |
| ០,២ ដល់ ០,៤ | ខ្សោយ |
| ០,៤ ដល់ ០,៦ | មធ្យម |
| ០,៦ ដល់ ០,៨ | ល្អ |
| ០,៨ ដល់ ១ | ល្អណាស់ |

ការសិក្សាវិទ្យាសាស្ត្រភាគច្រើន តម្រូវឲ្យរាយការណ៍ទាំង OA ទាំង Kappa រួមគ្នា ព្រោះ Kappa ទាប ខណៈ OA ខ្ពស់ ជាសញ្ញាព្រមានថាលទ្ធផលអាចត្រូវបានជំរុញដោយថ្នាក់ភាគច្រើនតែម្នាក់ឯង។

### ១៣.៣.៥. យុទ្ធសាស្ត្រយកគំរូផ្ទៀងផ្ទាត់

គុណភាពនៃការវាយតម្លៃ អាស្រ័យលើគុណភាពគំរូផ្ទៀងផ្ទាត់ ដូចគ្នានឹងគំរូបណ្ដុះបណ្ដាលដែរ។ គោលការណ៍ចាំបាច់៖

1. **ឯករាជ្យទាំងស្រុងពីគំរូបណ្ដុះបណ្ដាល**៖ កុំប្រើចំណុចដដែលសម្រាប់ទាំងពីរគោលបំណង។
2. **យកគំរូចៃដន្យ ឬចៃដន្យបែងចាត់ថ្នាក់ (stratified random)**៖ ចៃដន្យសុទ្ធអាចផ្ដល់ចំណុចតិចពេកសម្រាប់ថ្នាក់កម្រ (ដូចជាទឹក ដែលមានផ្ទៃតូច)។ ការបែងចាត់ថ្នាក់មុន រួចចៃដន្យក្នុងថ្នាក់នីមួយៗ ធានាថាថ្នាក់តូចមានគំរូគ្រប់គ្រាន់។
3. **ចំនួនគំរូគ្រប់គ្រាន់**៖ គោលការណ៍ណែនាំទូទៅ យ៉ាងតិច ៥០ ចំណុចក្នុងមួយថ្នាក់ សម្រាប់ការសិក្សាកម្រិតមធ្យម (ច្រើនជាងនេះសម្រាប់ការសិក្សាដែលទាមទារភាពត្រឹមត្រូវខ្ពស់ ឬតំបន់ធំ)។
4. **ប្រភពគំរូគុណភាពខ្ពស់ជាងទិន្នន័យដែលកំពុងវាយតម្លៃ**៖ ការចុះទីវាល ឬការបកស្រាយពីរូបភាពគុណភាពខ្ពស់ជាង (ឧ. ប្រើរូបភាពក្រឡា ០,៥ ម ដើម្បីផ្ទៀងផ្ទាត់ការចាត់ថ្នាក់ពី Sentinel-2)។

!!! warning "ភាពលំអៀងពេលវេលា"
    គំរូផ្ទៀងផ្ទាត់ដែលចុះទីវាលក្រោយការថតរូបភាពជាច្រើនខែ អាចលែងត្រឹមត្រូវទេ ជាពិសេសសម្រាប់ថ្នាក់ដែលប្ដូរលឿន (ដូចជាស្រែ)។ ត្រូវប្រុងប្រយ័ត្នអំពីភាពស្របគ្នានៃពេលវេលារវាងរូបភាព និងគំរូផ្ទៀងផ្ទាត់។

**ត្រូវការចំណុចផ្ទៀងផ្ទាត់ប៉ុន្មាន?** ទ្រឹស្ដីប្រូបាប៊ីលីតេទ្វេធាតុ (binomial) ផ្ដល់រូបមន្ត (Fitzpatrick-Lins, 1981)៖

> **N = Z² × p × q ÷ E²**

ដែល p = ភាពត្រឹមត្រូវដែលរំពឹង (%) · q = ១០០ − p · E = កំហុសដែលអនុញ្ញាត (%) · Z = ២ (កម្រិតទំនុកចិត្ត ~៩៥%)។

- p = ៨៥% · E = ៥% → N = ២² × ៨៥ × ១៥ ÷ ៥² = **២០៤** ចំណុច
- p = ៨៥% · E = ១០% → N = ២² × ៨៥ × ១៥ ÷ ១០² = **៥១** ចំណុច

ការកាត់បន្ថយកំហុសដែលអនុញ្ញាតពាក់កណ្ដាល ត្រូវការចំណុចបួនដង។ ចំនួននេះគឺសម្រាប់ផែនទីទាំងមូល ដែលត្រូវចែកទៅតាមថ្នាក់ (បែងចាត់ថ្នាក់)។

**ឧទាហរណ៍ KHAT ពេញលេញ**៖ រូបមន្តពេញ

> **K̂ = (N Σx<sub>ii</sub> − Σ(x<sub>i+</sub> × x<sub>+i</sub>)) ÷ (N² − Σ(x<sub>i+</sub> × x<sub>+i</sub>))**

ដែល x<sub>ii</sub> = អង្កត់ទ្រូង · x<sub>i+</sub> = ផលបូកជួរដេក · x<sub>+i</sub> = ផលបូកជួរឈរ។ តារាងមួយមាន N = ២៣០ · អង្កត់ទ្រូង ២៤ + ៣២ + ១៥ + ៣២ = ១០៣ (OA = ៤៥%) · Σ(x<sub>i+</sub> × x<sub>+i</sub>) = (៣៥×៦០) + (៩៩×៤៧) + (៣៦×៦៩) + (៦០×៥៤) = ១២ ៤៧៧។ ដូច្នេះ K̂ = (២៣០ × ១០៣ − ១២ ៤៧៧) ÷ (២៣០² − ១២ ៤៧៧) = ១១ ២១៣ ÷ ៤០ ៤២៣ = **០,២៨** ពោលគឺភាពត្រឹមត្រូវទាប។

**គណនាផ្ទៃពីលទ្ធផលចាត់ថ្នាក់**៖ ផ្ទៃថ្នាក់ = ចំនួនក្រឡា × ទំហំក្រឡា។ Landsat៖ ចំនួនក្រឡា × (៣០ × ៣០) ÷ (១០០ × ១០០) = ហិកតា។ ឧទាហរណ៍ ថ្នាក់ «ទឹក» មាន ១៦ ៤៥៨ ក្រឡា → ១៦ ៤៥៨ × ០,០៩ = **១ ៤៨១ ហ.ត**។ ចំណាំ៖ ផ្ទៃពីផែនទីរាប់ក្រឡាដែលចាត់ថ្នាក់ខុសផងដែរ ដូច្នេះការសិក្សាដែលល្អ កែតម្រូវផ្ទៃដោយប្រើតារាងច្របូកច្របល់ (Olofsson et al., 2014) ហើយរាយការណ៍ចន្លោះទំនុកចិត្ត។

### ១៣.៣.៦. ដែនកំណត់ និងអ្វីដែលការវាយតម្លៃមិនប្រាប់

1. **លេខភាពត្រឹមត្រូវសុពលភាពត្រឹមតែលើតំបន់សិក្សា**៖ ការចាត់ថ្នាក់ត្រឹមត្រូវ ៩០% លើខេត្តមួយ មិនមានន័យថានឹងត្រឹមត្រូវ ៩០% លើខេត្តផ្សេងទេ ជាពិសេសបើក្រប់ដីខុសគ្នា។
2. **ការវាយតម្លៃខ្លួនឯងអាចមានកំហុស**៖ បើគំរូផ្ទៀងផ្ទាត់មានកំហុស (ការកត់ត្រាខុស ការចុះទីវាលមិនត្រឹមត្រូវ) លទ្ធផលវាយតម្លៃខុសដែរ។
3. **ភាពត្រឹមត្រូវខ្ពស់មិនមានន័យថាព្រំដែនត្រឹមត្រូវទេ**៖ ចំណុចមួយចំនួនអាចចាត់ថ្នាក់ត្រូវ ខណៈព្រំរវាងថ្នាក់ (ដូចគែមព្រៃ) នៅតែមិនច្បាស់។
4. **ការវាយតម្លៃមិនប្រាប់មូលហេតុនៃកំហុសទេ**៖ វាប្រាប់ **ថាតើមានកំហុសប៉ុន្មាន** មិនមែន **ហេតុអ្វី** កើតកំហុសនោះទេ។ ការវិភាគតារាងច្របូកច្របល់ដោយប្រុងប្រយ័ត្ន (មើលថ្នាក់ណាច្រឡំនឹងថ្នាក់ណា) ជួយបង្កល័កណ្ឌនៃការកែលម្អ។

ការរាយការណ៍ត្រឹមត្រូវរួមមាន៖ តារាងច្របូកច្របល់ពេញលេញ · OA · Kappa · PA/UA តាមថ្នាក់ · ចំនួនគំរូ · និងវិធីយកគំរូ។

## ១៣.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ វាយតម្លៃ និងប្រៀបធៀបវិធីចាត់ថ្នាក់ទាំងពីរ

ត្រឡប់ទៅស្ថានភាពបើកមេរៀន៖ ប្រៀបធៀបលទ្ធផល K-means និងចម្ងាយអប្បបរមា ដោយប្រើគំរូផ្ទៀងផ្ទាត់ឯករាជ្យ ១៥០ ចំណុច (៣០ ក្នុងមួយថ្នាក់)។

### ជំហានទី១៖ បង្កើតតារាងច្របូកច្របល់ទាំងពីរ

សម្រាប់ការចាត់ថ្នាក់នីមួយៗ ប្រៀបធៀបលទ្ធផលនឹងក្រប់ដីពិតនៅ ១៥០ ចំណុច។

### ជំហានទី២៖ ប្រៀបធៀប OA

K-means (ក្រោយដាក់ស្លាក)៖ OA = ៨៦%។ ចម្ងាយអប្បបរមា (ជាមួយគំរូបណ្ដុះបណ្ដាល)៖ OA = ៩៣%។ ដំបូងមើលទៅ ចម្ងាយអប្បបរមាប្រសើរជាង។

### ជំហានទី៣៖ ប្រៀបធៀប Kappa

ព្រៃឈើគ្របដណ្ដប់ ៤៥% នៃទិន្នន័យ (ថ្នាក់ភាគច្រើន ប៉ុន្តែមិនលើសលប់)។ Kappa របស់ K-means = ០,៨០ ចម្ងាយអប្បបរមា = ០,៩០។ លំដាប់ដដែល ប៉ុន្តែគម្លាតតូចជាង OA ចង្អុលថាការឯកភាពដោយចៃដន្យមិនមែនជាកត្តាចម្បងទេ។

### ជំហានទី៤៖ វិភាគតារាងច្របូកច្របល់លម្អិត

ទាំងពីរវិធីមានបញ្ហាដូចគ្នា៖ ការច្រឡំរវាងព្រៃឈើ និងស្រែ (ដូចការសង្កេតក្នុង[មេរៀនទី៣](lesson-03.md))។ ចម្ងាយអប្បបរមាកែបញ្ហានេះកាន់តែប្រសើរ ដោយសារវាប្រើក្រុមរលក SWIR ដែលបែងចែកបានល្អជាង ខណៈ K-means ផ្អែកលើចម្ងាយសរុបគ្រប់ក្រុមរលកស្មើគ្នា។

### ជំហានទី៥៖ អនុសាសន៍

ចម្ងាយអប្បបរមាជាជម្រើសល្អជាងសម្រាប់ការសិក្សានេះ ព្រោះទាំង OA ទាំង Kappa ខ្ពស់ជាង ហើយភាពខុសគ្នាមិនមែនគ្រាន់តែជាផលនៃថ្នាក់ភាគច្រើនទេ។ ប៉ុន្តែក្រុមការងារគួររាយការណ៍ការច្រឡំព្រៃ/ស្រែជាដែនកំណត់រួមរបស់ទាំងពីរវិធី។

### សេចក្ដីសន្និដ្ឋានដែលសមស្រប

> «ការវាយតម្លៃភាពត្រឹមត្រូវជាផ្លូវការ បង្ហាញថាចម្ងាយអប្បបរមាប្រសើរជាង K-means សម្រាប់ការសិក្សានេះ ដោយផ្អែកលើទាំង OA ទាំង Kappa មិនមែនការសង្កេតដោយភ្នែកតែម្នាក់ឯងទេ។ ការវិភាគតារាងច្របូកច្របល់លម្អិត ក៏បញ្ជាក់ថាបញ្ហាចម្បង (ការច្រឡំព្រៃ/ស្រែ) នៅតែមានក្នុងទាំងពីរវិធី។»

!!! warning "អ្វីដែលយើងមិនអាចសន្និដ្ឋាន"
    ការវាយតម្លៃនេះសុពលភាពត្រឹមតែលើតំបន់សិក្សា និងពេលវេលាដែលប្រើ។ វាមិនអាចធានាថាចម្ងាយអប្បបរមានឹងប្រសើរជាង K-means សម្រាប់តំបន់ ឬឆ្នាំផ្សេង។ ការធ្វើផែនទីលើតំបន់ថ្មី គួរធ្វើការវាយតម្លៃម្ដងទៀត។

## ១៣.៥. សកម្មភាពអនុវត្ត

!!! tip "លំហាត់នេះស្ថិតក្នុងសៀវភៅអនុវត្ត"
    ការអនុវត្តលើ QGIS សម្រាប់មេរៀននេះស្ថិតក្នុង [**លំហាត់ទី១៣**](../workbook/lab-13.md)។

    និស្សិតធ្វើលំហាត់នេះ **ដោយខ្លួនឯង** នៅក្រៅម៉ោងរៀន។

## ១៣.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន

**១. «ភាពត្រឹមត្រូវសរុបខ្ពស់ គឺគ្រប់គ្រាន់ដើម្បីជឿទុកចិត្តលើផែនទី»**
Kappa និង PA/UA តាមថ្នាក់ ប្រាប់ព័ត៌មានបន្ថែមសំខាន់ដែល OA លាក់។

**២. «Producer's និង User's Accuracy ដូចគ្នា»**
ពួកវាគណនាពីទិសដៅផ្ទុយគ្នា (ជួរដេក ធៀបនឹងជួរឈរ) ហើយអាចខុសគ្នាខ្លាំងសម្រាប់ថ្នាក់តែមួយ។

**៣. «Kappa ១ តែងប្រសើរជាង Kappa ០,៨»**
តម្លៃខ្ពស់ជាងតែងល្អជាង ប៉ុន្តែភាពខុសគ្នាតូចអាចមិនមានន័យស្ថិតិ បើគំរូតិច។

**៤. «គំរូបណ្ដុះបណ្ដាល និងគំរូផ្ទៀងផ្ទាត់អាចជាសំណុំដូចគ្នា ដរាបណាមានច្រើនគ្រប់គ្រាន់»**
ការប្រើគំរូដដែលផ្ដល់ការវាយតម្លៃក្លែងក្លាយ ទោះមានចំនួនប៉ុន្មានក៏ដោយ។

**៥. «ភាពត្រឹមត្រូវខ្ពស់លើគំរូមានន័យថាព្រំដែនផែនទីត្រឹមត្រូវ»**
ចំណុចគំរូវាស់តែចំណុចនោះ។ ព្រំដែនរវាងចំណុច (ដូចគែមព្រៃ) នៅតែអាចមិនច្បាស់។

## ១៣.៧. សេចក្ដីសង្ខេបមេរៀន

តារាងច្របូកច្របល់ប្រៀបធៀបក្រប់ដីព្យាករជាមួយក្រប់ដីពិត ដែលផ្ដល់មូលដ្ឋានសម្រាប់គណនា Overall Accuracy (ជារួម) Producer's Accuracy (ការខកខាន តាមថ្នាក់) និង User's Accuracy (ការបន្ថែមខុស តាមថ្នាក់)។

Kappa Coefficient កែតម្រូវសម្រាប់ការឯកភាពដោយចៃដន្យ ដែលសំខាន់ពេលថ្នាក់មួយគ្របដណ្ដប់ភាគច្រើននៃទិន្នន័យ។ គុណភាពនៃការវាយតម្លៃ អាស្រ័យលើគំរូផ្ទៀងផ្ទាត់ដែលឯករាជ្យ ចែកចាយសមស្រប និងមានគុណភាព។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    ភាពត្រឹមត្រូវមិនមែនជាលេខតែមួយទេ។ តារាងច្របូកច្របល់ពេញលេញប្រាប់រឿងច្រើនជាង OA តែម្នាក់ឯង ជាពិសេសថាតើកំហុសនៅឯណា។

## ១៣.៩. សំណួររំលឹក និងវាយតម្លៃ

### ក. សំណួរយល់ដឹង

1. សរសេររូបមន្ត Overall Accuracy និងពន្យល់ព័ត៌មានដែលវាលាក់។
2. បែងចែក Producer's Accuracy និង User's Accuracy ព្រមទាំងឧទាហរណ៍។
3. ហេតុអ្វី Kappa Coefficient ចាំបាច់ ក្រៅពី OA?
4. រាយគោលការណ៍បួននៃការយកគំរូផ្ទៀងផ្ទាត់ដែលមានគុណភាព។
5. រាយដែនកំណត់បីនៃការវាយតម្លៃភាពត្រឹមត្រូវ។

### ខ. សំណួរអនុវត្តគោលគំនិត

6. ថ្នាក់ «ទឹក» មាន PA=៩៥% និង UA=៦០%។ ពន្យល់ន័យ ហើយស្នើកន្លែងដែលបញ្ហាទំនងជាកើត។
7. ការសិក្សាមួយរាយការណ៍ OA=៩២% ដោយមិនរាយ Kappa។ តើអ្នកនឹងសួរអ្វីមុននឹងជឿទុកចិត្តលទ្ធផល?
8. អ្នកមានតំបន់សិក្សាដែលមាន ៩០% ជាស្រែ និង ១០% ជាថ្នាក់ផ្សេង។ ស្នើយុទ្ធសាស្ត្រយកគំរូផ្ទៀងផ្ទាត់ដែលសមស្រប។
9. គណនា OA ដោយដៃ បើតារាងច្របូកច្របល់ ២ × ២ មាន [៨០,៥] / [១០,៦៥] (ជួរដេក/ជួរឈរ)។
10. ហេតុអ្វីភាពត្រឹមត្រូវខ្ពស់នៅចំណុចគំរូ មិនធានាថាព្រំដែនផែនទីទាំងមូលត្រឹមត្រូវ? ផ្ដល់ឧទាហរណ៍មួយ។

### គ. កិច្ចការខ្លី

ក្នុងពិសោធន៍តារាងច្របូកច្របល់ ចូរបង្កើតតារាងជាមួយគំរូ ៤០ ក្នុងមួយថ្នាក់។ គណនាដោយដៃ (លើក្រដាស)៖ OA · PA និង UA សម្រាប់ថ្នាក់ «ស្រែ» ដោយប្រើលេខពីតារាង។ ប្រៀបធៀបចម្លើយរបស់អ្នក ជាមួយលេខដែលពិសោធន៍បង្ហាញ។
