# មេរៀនទី១២៖ ការចាត់ថ្នាក់មានការណែនាំ

!!! info "ព័ត៌មានមេរៀន"
    **រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង** · មេរៀនទី ១២ ក្នុងចំណោម ១៥

ការចាត់ថ្នាក់គ្មានការណែនាំ ([មេរៀនទី១១](lesson-11.md)) មានចំណុចខ្សោយមួយធំ៖ វាមិនដឹងថាចង្កោមនីមួយៗគឺជាអ្វីទេ ហើយអ្នកវិភាគត្រូវដាក់ស្លាកក្រោយការគណនា។ បើអ្នកមានចំណេះដឹងស្រាប់ថាកន្លែងណាជាព្រៃ កន្លែងណាជាស្រែ ហេតុអ្វីមិនប្រើចំណេះដឹងនោះជា **ព័ត៌មានបញ្ចូល** ដល់ក្បួនដោះស្រាយតាំងពីដើម?

នេះជាគំនិតស្នូលនៃ **ការចាត់ថ្នាក់មានការណែនាំ (Supervised Classification)**៖ អ្នកបង្ហាញឧទាហរណ៍ (គំរូបណ្ដុះបណ្ដាល) ដល់ក្បួនដោះស្រាយ រួចវារៀនលក្ខណៈស្ពិចត្រាល់ពីឧទាហរណ៍នោះ ដើម្បីចាត់ថ្នាក់ផ្នែកនៅសល់នៃរូបភាព។

---

## ១២.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. ពន្យល់ដំណើរការទាំងបួនជំហាននៃការចាត់ថ្នាក់មានការណែនាំ។
2. ជ្រើស និងវាយតម្លៃគំរូបណ្ដុះបណ្ដាលប្រកបដោយគុណភាព។
3. ពន្យល់ក្បួនដោះស្រាយចម្ងាយអប្បបរមា និង Parallelepiped ជាមូលដ្ឋាន។
4. ប្រៀបធៀបក្បួនដោះស្រាយសាមញ្ញ ជាមួយក្បួនដោះស្រាយទំនើបជាងដូច Random Forest។
5. កំណត់កំហុសទូទៅក្នុងការជ្រើសគំរូបណ្ដុះបណ្ដាល។

## ១២.២. ស្ថានភាពបើកមេរៀន៖ ប្រើចំណេះដឹងដែលមានស្រាប់

ក្រុមការងារដដែលពីមេរៀនមុន ឥឡូវមានពេលបន្ថែម និងអាចចុះទីវាលបានបន្តិច។ ពួកគេចុះទៅ ២០ ចំណុច កត់ត្រាកូអរដោនេ GPS ព្រមព័ត៌មានគម្របដីពិត (ទឹក ព្រៃ ស្រែ សំណង់ ដីទទេ)។

ជំនួសឲ្យប្រើ K-means ដែលមិនដឹងអត្ថន័យចង្កោម ពួកគេចង់ប្រើចំណុចទាំង ២០ នេះ ដើម្បី **បង្រៀន** ក្បួនដោះស្រាយថាតម្លៃក្រុមរលកបែបណាត្រូវនឹងគម្របដីមួយណា។ រួចប្រើចំណេះដឹងនោះចាត់ថ្នាក់ក្រឡាទាំងអស់ដែលនៅសល់ក្នុងរូបភាព ដែលមានច្រើនជាងចំណុចគំរូរាប់ពាន់ដង។

សំណួរដែលមេរៀននេះឆ្លើយ៖ តើត្រូវយកគំរូបណ្ដុះបណ្ដាលយ៉ាងណា ដើម្បីឲ្យវាមានគុណភាព? តើក្បួនដោះស្រាយចាត់ថ្នាក់ក្រឡាថ្មីដោយរបៀបណា? ហើយហេតុអ្វីវិធីនេះជាទូទៅផ្ដល់ភាពត្រឹមត្រូវខ្ពស់ជាងការចាត់ថ្នាក់គ្មានការណែនាំ?

---

## ១២.៣. ទ្រឹស្ដីស្នូល

### ១២.៣.១. ដំណើរការទាំងបួនជំហាន

<figure markdown>
--8<-- "assets/svg/r12-workflow.svg"
<figcaption>រូបទី១២.១៖ ដំណើរការនៃការចាត់ថ្នាក់មានការណែនាំ។ មនុស្សផ្ដល់គំរូដែលដឹងស្រាប់ថាជាប្រភេទអ្វី រួចក្បួនដោះស្រាយរៀនពីគំរូនោះ ដើម្បីចាត់ថ្នាក់ផ្នែកនៅសល់។</figcaption>
</figure>

**១. យកគំរូបណ្ដុះបណ្ដាល (Training samples)**៖ ជ្រើសក្រឡាដែលដឹងច្បាស់ថាជាគម្របដីអ្វី ដោយប្រើចំណុច GPS ទីវាល ការបកស្រាយដោយភ្នែកលើរូបភាពគុណភាពខ្ពស់ ឬចំណេះដឹងតំបន់។

**២. គណនាសញ្ញាណស្ពិចត្រាល់សម្រាប់ថ្នាក់នីមួយៗ**៖ សម្រាប់ថ្នាក់នីមួយៗ គណនាតម្លៃមធ្យម (និងជួនកាលគម្លាតគំរូ) នៃក្រុមរលកទាំងអស់ ពីគំរូបណ្ដុះបណ្ដាលរបស់ថ្នាក់នោះ។

**៣. ចាត់ថ្នាក់ក្រឡាទាំងអស់**៖ សម្រាប់ក្រឡានីមួយៗក្នុងរូបភាព (រួមទាំងក្រឡាដែលមិនមែនជាគំរូ) គណនាថាតម្លៃរបស់វាជិតនឹងសញ្ញាណថ្នាក់ណាបំផុត រួចចាត់ថ្នាក់តាមនោះ។

**៤. វាយតម្លៃភាពត្រឹមត្រូវ**៖ ប្រើគំរូឯករាជ្យ (មិនប្រើក្នុងជំហានទី១) ដើម្បីពិនិត្យថាការចាត់ថ្នាក់ត្រឹមត្រូវប៉ុនណា ([មេរៀនទី១៣](lesson-13.md))។

ភាពខុសគ្នាស្នូលពី K-means៖ ការចាត់ថ្នាក់មានការណែនាំ **ចាប់ផ្ដើមពីចំណេះដឹងរបស់មនុស្ស** មិនមែនពីទិន្នន័យតែម្នាក់ឯងទេ។

### ១២.៣.២. ក្បួនដោះស្រាយចម្ងាយអប្បបរមា

ក្បួនដោះស្រាយសាមញ្ញបំផុតគឺ **ចម្ងាយអប្បបរមាទៅកណ្ដាលថ្នាក់ (Minimum Distance to Mean)**៖ សម្រាប់ក្រឡានីមួយៗ គណនាចម្ងាយ (ក្នុងលំហលក្ខណៈ) ទៅកណ្ដាលនៃថ្នាក់នីមួយៗ រួចចាត់ថ្នាក់ទៅថ្នាក់ដែលមានចម្ងាយខ្លីបំផុត។ វិធីនេះស្រដៀងនឹងជំហាន «ចាត់តាំង» នៃ K-means ប៉ុន្តែកណ្ដាលថ្នាក់មកពីគំរូដែលមនុស្សកំណត់ មិនមែនរកដោយចៃដន្យទេ។

!!! example "ពិសោធន៍៖ ចាត់ថ្នាក់ដោយចម្ងាយអប្បបរមា"
    ប្ដូរចំនួនគំរូក្នុងមួយថ្នាក់ ហើយចុច «យកគំរូ និងចាត់ថ្នាក់»។ សង្កេតថាភាពត្រឹមត្រូវទំនោរកើនឡើងជាមួយចំនួនគំរូបន្ថែម ប៉ុន្តែមិនស្មើ ១០០% ជានិច្ចទេ។

<div class="sim" data-sim="rs-supervised"></div>

ចំណុចខ្សោយចម្បងនៃវិធីនេះ៖ **ក្រឡាគ្រប់ក្រឡាត្រូវបានចាត់ថ្នាក់ជានិច្ច** សូម្បីតែក្រឡាដែលមិនស្រដៀងថ្នាក់ណាមួយសោះ (ដូចជាដំបូលពណ៌ចម្លែក ឬវត្ថុថ្មីដែលគ្មានក្នុងគំរូ)។ វាចាត់ចូលទៅថ្នាក់ដែល «ជិតបំផុត» ទោះជិតបំផុតនោះនៅឆ្ងាយក៏ដោយ។

### ១២.៣.៣. Parallelepiped៖ ជម្រើសដែលអនុញ្ញាតឲ្យ «មិនដឹង»

ក្បួនដោះស្រាយមួយទៀតគឺ **Parallelepiped**៖ សម្រាប់ថ្នាក់នីមួយៗ កំណត់ **ជួរតម្លៃ** (អប្បបរមា/អតិបរមា ឬមធ្យម ± គម្លាតគំរូ) សម្រាប់ក្រុមរលកនីមួយៗ ដោយផ្អែកលើគំរូបណ្ដុះបណ្ដាល ដែលបង្កើត «ប្រអប់» ក្នុងលំហលក្ខណៈ។ ក្រឡាមួយត្រូវបានចាត់ថ្នាក់ទៅថ្នាក់ណាដែលតម្លៃរបស់វាធ្លាក់ក្នុងប្រអប់នោះ។

!!! example "ពិសោធន៍៖ ព្រំដែនការសម្រេចចិត្ត"
    ប្ដូររវាងវិធីទាំងពីរ ហើយសង្កេតតំបន់ពណ៌ស (មិនបានចាត់ថ្នាក់) ក្នុងវិធី Parallelepiped។

<div class="sim" data-sim="rs-boundary"></div>

ភាពខុសគ្នាសំខាន់៖ Parallelepiped អនុញ្ញាតឲ្យមាន **ក្រឡាមិនបានចាត់ថ្នាក់ (unclassified)** បើតម្លៃរបស់វាមិនធ្លាក់ក្នុងប្រអប់ណាមួយ។ វាស្មោះត្រង់ជាង (មិនធ្វើពុតថាដឹង ពេលមិនប្រាកដ) ប៉ុន្តែផលិតលទ្ធផលដែលមានចន្លោះទទេ ដែលអ្នកប្រើមួយចំនួនចាត់ទុកថាមិនងាយស្រួល។

ក្បួនដោះស្រាយទាំងពីរនេះ (ចម្ងាយអប្បបរមា និង Parallelepiped) សាមញ្ញ និងងាយពន្យល់ ដែលជាមូលហេតុដែលវាល្អសម្រាប់ការសិក្សាគោលការណ៍មូលដ្ឋាន។ ក្នុងការអនុវត្តជាក់ស្ដែងទំនើប ក្បួនដោះស្រាយស្មុគស្មាញជាងតែងផ្ដល់ភាពត្រឹមត្រូវខ្ពស់ជាង។

| ក្បួនដោះស្រាយ | គោលការណ៍សង្ខេប | គុណសម្បត្តិ |
|---|---|---|
| **Maximum Likelihood** | គណនាប្រូបាប៊ីលីតេស្ថិតិដោយសន្មតការចែកចាយធម្មតា | គិតគូរទាំងរូបរាង និងទំហំនៃចង្កោមថ្នាក់ |
| **Random Forest** | ដើមឈើសម្រេចចិត្តជាច្រើន បោះឆ្នោតរួមគ្នា | ត្រឹមត្រូវខ្ពស់ ធន់នឹងសំឡេងរំខាន ប្រើក្រុមរលកច្រើនបានល្អ |
| **Support Vector Machine (SVM)** | រកបន្ទាត់ (ឬផ្ទៃ) ដែលបំបែកថ្នាក់បានល្អបំផុត | ដំណើរការល្អជាមួយគំរូតិច |
| **Neural Network / Deep Learning** | ស្រទាប់គណនាច្រើនរៀនលក្ខណៈស្មុគស្មាញ | ត្រឹមត្រូវបំផុតជាមួយទិន្នន័យច្រើន ប៉ុន្តែត្រូវការគំរូច្រើន និងកុំព្យូទ័រខ្លាំង |

ក្នុង QGIS **Semi-Automatic Classification Plugin (SCP)** ផ្ដល់ជម្រើសចម្ងាយអប្បបរមា Maximum Likelihood និង Spectral Angle Mapping។ សម្រាប់ Random Forest ឬវិធីទំនើបជាង ត្រូវប្រើកម្មវិធីជំនួយ **Orfeo Toolbox (OTB)** ដែលភ្ជាប់ជាមួយ QGIS ឬប្រើ Python/scikit-learn ដោយផ្ទាល់ (ជាខ្លឹមសារកម្រិតខ្ពស់ជាងវិសាលភាពសៀវភៅនេះ)។

### ១២.៣.៥. ការជ្រើសគំរូបណ្ដុះបណ្ដាលប្រកបដោយគុណភាព

គុណភាពនៃលទ្ធផលចាត់ថ្នាក់ **អាស្រ័យលើគុណភាពនៃគំរូបណ្ដុះបណ្ដាល ច្រើនជាងក្បួនដោះស្រាយ**។ គោលការណ៍ណែនាំ៖

1. **ចំនួនគំរូគ្រប់គ្រាន់**៖ យ៉ាងតិច ១០ ដល់ ៣០ ក្រឡាក្នុងមួយថ្នាក់ តាមចំនួនក្រុមរលក (ច្រើនក្រុមរលក ត្រូវការគំរូច្រើនជាង)។
2. **ភាពបរិសុទ្ធ**៖ ជៀសវាងក្រឡាព្រំដែន ឬក្រឡាចម្រុះ។ ជ្រើសកណ្ដាលនៃផ្ទៃធំគ្រប់គ្រាន់នៃថ្នាក់នីមួយៗ។
3. **ការចែកចាយសព្វទូទាំងតំបន់**៖ គំរូដែលប្រមូលផ្ដុំតែជ្រុងមួយ អាចមិនតំណាងឲ្យប្រែប្រួលនៃថ្នាក់នោះទូទាំងតំបន់ (ឧ. ព្រៃនៅជម្រាលខុសគ្នា អាចមានស្រមោលខុសគ្នា)។
4. **តំណាងឲ្យប្រែប្រួលខាងក្នុងថ្នាក់**៖ «ស្រែ» រួមទាំងស្រែខ្ចី និងស្រែទុំ។ បើគំរូយកតែស្រែខ្ចី ក្បួនដោះស្រាយនឹងខកខានចាត់ថ្នាក់ស្រែទុំបានត្រឹមត្រូវ។
5. **ឯករាជ្យពីគំរូវាយតម្លៃ**៖ គំរូសម្រាប់ជំហានទី១ (បណ្ដុះបណ្ដាល) និងជំហានទី៤ (វាយតម្លៃ) ត្រូវខុសគ្នា ជៀសវាងលទ្ធផលដែលមើលទៅល្អហួសហេតុ។

!!! warning "កំហុសទូទៅបំផុត"
    កំហុសទូទៅបំផុតរបស់និស្សិត គឺប្រើគំរូតែមួយសំណុំសម្រាប់ទាំងបណ្ដុះបណ្ដាល ទាំងវាយតម្លៃ។ នេះផ្ដល់ភាពត្រឹមត្រូវដែលមើលទៅខ្ពស់ណាស់ ប៉ុន្តែជា **ការវាស់ក្លែងក្លាយ** ព្រោះក្បួនដោះស្រាយ «ធ្លាប់ឃើញ» ទិន្នន័យទាំងនោះរួចមកហើយ។ លំហាត់ទី១៣ នឹងបង្ហាញពីរបៀបញែកគំរូឲ្យត្រឹមត្រូវ។

### ១២.៣.៦. ក្បួនជាក់ស្ដែងសម្រាប់តំបន់គំរូ (ROI)

| លក្ខណៈវិនិច្ឆ័យ | ក្បួន |
|---|---|
| **ចំនួនក្រឡា** | ក្រុមរលក n → យ៉ាងតិច **១០n ក្រឡា** ក្នុងមួយថ្នាក់ (Landsat ៦ ក្រុមរលក → > ៦០ ក្រឡា) |
| **ចំនួនតំបន់** | យ៉ាងតិច **៥ ដល់ ១០ តំបន់** ក្នុងមួយថ្នាក់ |
| **ទីតាំង** | ចែកសព្វរូបភាព មិនប្រមូលផ្ដុំកន្លែងតែមួយ |
| **ទំហំ** | ធំល្មមដើម្បីប៉ាន់ស្មានមធ្យម និងគម្លាតគំរូបានត្រឹមត្រូវ |
| **ភាពឯកសណ្ឋាន** | អ៊ីស្តូក្រាមក្នុងក្រុមរលកនីមួយៗមានកំពូលតែមួយ (unimodal) |

**មូលហេតុនៃការច្រឡំថ្នាក់**៖ គម្របដីមានសញ្ញាណស្រដៀងគ្នា (ដំណាំផ្សេងៗ ដីទទេ និងសំណង់) · ស្រមោល ឬពពក · តំបន់គំរូគូសធំពេក ធ្វើឲ្យរួមបញ្ចូលគម្របដីច្រើនប្រភេទ ឬតូចពេក មិនចាប់បានភាពប្រែប្រួលក្នុងថ្នាក់។ ដំណោះស្រាយ៖ បង្កើតតំបន់គំរូច្រើនតូចៗ សម្រាប់ផ្នែកផ្សេងៗនៃថ្នាក់តែមួយ ជាជាងតំបន់ធំមួយ។

**គូសពហុកោណ ធៀបនឹង Region Growing**៖ សម្រាប់ផ្ទៃធំឯកសណ្ឋាន (បឹង ទន្លេ) គូសពហុកោណដោយដៃបាន។ សម្រាប់ផ្ទៃដែលប្រែប្រួលច្រើន (ព្រៃលើភ្នំ ចម្ការ) ប្រើ **Region Growing**៖ ចាប់ផ្ដើមពីក្រឡាគ្រាប់ពូជ (seed) រួមបញ្ចូលក្រឡាជិតខាងដែលមាន **ចម្ងាយស្ពិចត្រាល់** តិចជាងកម្រិតកំណត់ (ឧ. ០,០៨ សម្រាប់ទិន្នន័យចាំងផ្លាត ០–១) រហូតដល់ទំហំអតិបរមា។

**Maximum Likelihood និង Spectral Angle Mapper**៖

- **ចម្ងាយអប្បបរមា** ប្រើតែ **មធ្យម** នៃថ្នាក់។
- **Maximum Likelihood** ប្រើទាំង **មធ្យម និងបម្រែបម្រួល (covariance)** ដើម្បីគណនាប្រូបាប៊ីលីតេដែលក្រឡាមួយជាថ្នាក់នីមួយៗ ដោយសន្មតការចែកចាយធម្មតា។ ថ្នាក់ដែលមានបម្រែបម្រួលធំ (ទីក្រុង) «ចាប់» ក្រឡាបានទូលាយជាងថ្នាក់ដែលឯកសណ្ឋាន (ទឹក)។ ត្រូវការគំរូច្រើនជាងចម្ងាយអប្បបរមា ដើម្បីប៉ាន់ស្មាន covariance បានត្រឹមត្រូវ។
- **Spectral Angle Mapper (SAM)** វាស់ **មុំ** រវាងវ៉ិចទ័រស្ពិចត្រាល់ មិនមែនចម្ងាយទេ ដូច្នេះវាធន់នឹងភាពខុសគ្នានៃពន្លឺ (ជម្រាលភ្នំដែលមានស្រមោល និងមុខដែលត្រូវពន្លឺ មានរាងខ្សែដូចគ្នា ប៉ុន្តែកម្រិតខុសគ្នា)។

**Macroclass និង Class ក្នុង SCP**៖ SCP អនុញ្ញាតឲ្យគូសតំបន់គំរូជា **class** លម្អិត (ស្មៅ ដើមឈើ អគារ ផ្លូវ) ហើយដាក់ពួកវាក្រោម **macroclass** ធំ (រុក្ខជាតិ តំបន់សាងសង់)។ ការចាត់ថ្នាក់តាម class ធ្វើឲ្យគំរូឯកសណ្ឋាន ហើយលទ្ធផលចុងក្រោយអាចបង្ហាញតាម macroclass។

## ១២.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ ចាត់ថ្នាក់ដោយប្រើគំរូទីវាល ២០ ចំណុច

ក្រុមការងារមានចំណុច GPS ២០ ដែលបានពិនិត្យទីវាល ចែកទៅតាមថ្នាក់ប្រាំ (៤ ចំណុចក្នុងមួយថ្នាក់)។

### ជំហានទី១៖ វាយតម្លៃគំរូ

៤ ចំណុចក្នុងមួយថ្នាក់ តិចជាងអប្បបរមា ១០ ដែលបានណែនាំ។ ក្រុមការងារសម្រេចចិត្តបន្ថែមគំរូពីការបកស្រាយដោយភ្នែក (មិនចុះទីវាល) លើផ្ទៃធំច្បាស់ៗ ដើម្បីបានយ៉ាងតិច ១៥ ក្រឡាក្នុងមួយថ្នាក់។

### ជំហានទី២៖ គណនាសញ្ញាណ

សម្រាប់ថ្នាក់នីមួយៗ គណនាតម្លៃមធ្យមក្រុមរលកទាំងប្រាំមួយ (B2 B3 B4 B8 B11 B12) ពីគំរូបណ្ដុះបណ្ដាល។

### ជំហានទី៣៖ ចាត់ថ្នាក់

ប្រើវិធីចម្ងាយអប្បបរមា ចាត់ថ្នាក់ក្រឡាទាំងអស់ក្នុងតំបន់សិក្សា (ច្រើនជាងគំរូរាប់ម៉ឺនដង)។

### ជំហានទី៤៖ ពិនិត្យដោយភ្នែក

ត្រួតលទ្ធផលលើពណ៌ពិត។ ភាគច្រើនស៊ីគ្នាល្អ ប៉ុន្តែមានតំបន់ព្រំដងទន្លេ ដែលចាត់ថ្នាក់ខុសទៅជា «សំណង់» ដោយសារជាក្រឡាចម្រុះទឹក/ដី។

### ជំហានទី៥៖ សម្គាល់សម្រាប់ជំហានបន្ទាប់

កត់ត្រាតំបន់ដែលមានបញ្ហា ជាការត្រៀមសម្រាប់ការវាយតម្លៃភាពត្រឹមត្រូវជាផ្លូវការ ([មេរៀនទី១៣](lesson-13.md)) ដែលនឹងផ្ដល់លេខច្បាស់លាស់ជាងការសង្កេតដោយភ្នែក។

### សេចក្ដីសន្និដ្ឋានដែលសមស្រប

> «ការចាត់ថ្នាក់មានការណែនាំប្រើប្រាស់ចំណេះដឹងទីវាលប្រកបដោយប្រសិទ្ធភាព ដោយពង្រីកចំណុចគំរូតិចទៅជាផែនទីពេញលេញ។ គុណភាពលទ្ធផលអាស្រ័យខ្លាំងលើគុណភាព និងបរិមាណនៃគំរូបណ្ដុះបណ្ដាល។»

!!! warning "អ្វីដែលយើងមិនអាចសន្និដ្ឋាន"
    ការពិនិត្យដោយភ្នែកមិនអាចផ្ដល់តួលេខភាពត្រឹមត្រូវជាផ្លូវការបានទេ។ វាអាចរកឃើញតែបញ្ហាធំៗដែលមើលឃើញច្បាស់ តំបន់ដែលមានកំហុសតូចៗអាចនៅមិនត្រូវបានរកឃើញ រហូតដល់ការវាយតម្លៃជាផ្លូវការ។

## ១២.៥. សកម្មភាពអនុវត្ត

!!! tip "លំហាត់នេះស្ថិតក្នុងសៀវភៅអនុវត្ត"
    ការអនុវត្តលើ QGIS សម្រាប់មេរៀននេះស្ថិតក្នុង [**លំហាត់ទី១២**](../workbook/lab-12.md)។

    និស្សិតធ្វើលំហាត់នេះ **ដោយខ្លួនឯង** នៅក្រៅម៉ោងរៀន។

## ១២.៦. ការយល់ច្រឡំដែលត្រូវប្រុងប្រយ័ត្ន

**១. «ការចាត់ថ្នាក់មានការណែនាំតែងត្រឹមត្រូវជាងគ្មានការណែនាំ»**
វាជាទូទៅត្រឹមត្រូវជាង ប៉ុន្តែអាស្រ័យខ្លាំងលើគុណភាពគំរូ។ គំរូមិនល្អ អាចផ្ដល់លទ្ធផលអាក្រក់ជាង K-means ដែលធ្វើឡើងដោយប្រុងប្រយ័ត្ន។

**២. «ចំនួនគំរូច្រើនជានិច្ចប្រសើរជាង»**
ចំនួនច្រើនល្អ ប៉ុន្តែគុណភាព (ភាពបរិសុទ្ធ ការតំណាង) សំខាន់ជាងចំនួនតែម្នាក់ឯង។

**៣. «Parallelepiped ធ្វើឲ្យក្រឡាទាំងអស់ត្រូវបានចាត់ថ្នាក់ដូចចម្ងាយអប្បបរមា»**
Parallelepiped អនុញ្ញាតឲ្យមានក្រឡាមិនបានចាត់ថ្នាក់ ចំណែកចម្ងាយអប្បបរមាចាត់ថ្នាក់គ្រប់ក្រឡាជានិច្ច។

**៤. «ការប្រើគំរូតែមួយសំណុំសម្រាប់បណ្ដុះបណ្ដាល និងវាយតម្លៃ គឺមិនអីទេ ដរាបណាមានច្រើន»**
នេះជាកំហុសសំខាន់ដែលផ្ដល់ភាពត្រឹមត្រូវក្លែងក្លាយ។ ត្រូវញែកគំរូជាពីរសំណុំឯករាជ្យ។

**៥. «Random Forest និងក្បួនដោះស្រាយទំនើបតែងជាជម្រើសល្អបំផុត»**
ពួកវាត្រូវការគំរូច្រើន និងកុំព្យូទ័រខ្លាំងជាង។ ចំពោះគម្រោងតូច ឬគំរូតិច វិធីសាមញ្ញអាចគ្រប់គ្រាន់ និងងាយបកស្រាយជាង។

## ១២.៧. សេចក្ដីសង្ខេបមេរៀន

ការចាត់ថ្នាក់មានការណែនាំចាប់ផ្ដើមពីគំរូបណ្ដុះបណ្ដាលដែលមនុស្សកំណត់ គណនាសញ្ញាណស្ពិចត្រាល់សម្រាប់ថ្នាក់នីមួយៗ រួចចាត់ថ្នាក់ក្រឡាទាំងអស់ដោយប្រៀបធៀបនឹងសញ្ញាណនោះ។

ចម្ងាយអប្បបរមាចាត់ថ្នាក់ក្រឡាគ្រប់ក្រឡាទៅថ្នាក់ជិតបំផុត ចំណែក Parallelepiped អនុញ្ញាតឲ្យមានក្រឡាមិនបានចាត់ថ្នាក់។ ក្បួនដោះស្រាយទំនើបជាង (Random Forest SVM) ជាទូទៅផ្ដល់ភាពត្រឹមត្រូវខ្ពស់ជាង។ គុណភាពគំរូបណ្ដុះបណ្ដាល (ចំនួន ភាពបរិសុទ្ធ ការតំណាង) សំខាន់ជាងការជ្រើសក្បួនដោះស្រាយ។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    ក្បួនដោះស្រាយគ្រាន់តែពង្រីកចំណេះដឹងដែលអ្នកផ្ដល់ឲ្យវា។ គំរូបណ្ដុះបណ្ដាលមិនល្អ ផ្ដល់ផែនទីមិនល្អ ទោះជាប្រើក្បួនដោះស្រាយទំនើបប៉ុនណាក៏ដោយ។

## ១២.៩. សំណួររំលឹក និងវាយតម្លៃ

### ក. សំណួរយល់ដឹង

1. រាយជំហានទាំងបួននៃការចាត់ថ្នាក់មានការណែនាំ។
2. ភាពខុសគ្នារវាងចម្ងាយអប្បបរមា និង Parallelepiped គឺអ្វី?
3. ហេតុអ្វីមិនគួរប្រើគំរូតែមួយសំណុំសម្រាប់ទាំងបណ្ដុះបណ្ដាល ទាំងវាយតម្លៃ?
4. រាយលក្ខណៈបួននៃគំរូបណ្ដុះបណ្ដាលដែលមានគុណភាព។
5. ប្រៀបធៀប Random Forest ជាមួយចម្ងាយអប្បបរមា តាមលក្ខណៈវិនិច្ឆ័យពីរ។

### ខ. សំណួរអនុវត្តគោលគំនិត

6. អ្នកមានគំរូតែ ៣ ក្រឡាក្នុងមួយថ្នាក់ដោយសារពេលកំណត់។ ស្នើវិធីបំពេញបន្ថែមដោយមិនចុះទីវាលបន្ថែម។
7. ការចាត់ថ្នាក់មួយបង្ហាញថាតំបន់ដីទទេទាំងអស់ត្រូវបានចាត់ថ្នាក់ខុសទៅជាសំណង់។ ស្នើមូលហេតុ និងដំណោះស្រាយ។
8. ហេតុអ្វី Parallelepiped អាចទុកតំបន់ធំមួយជា «មិនបានចាត់ថ្នាក់» ខណៈចម្ងាយអប្បបរមាមិនដែលទុកទទេ?
9. អ្នកមានតែ QGIS ជាមួយ SCP (គ្មាន Orfeo Toolbox)។ តើអ្នកអាចប្រើក្បួនដោះស្រាយអ្វីខ្លះ?
10. ស្នើលក្ខណៈវិនិច្ឆ័យបីសម្រាប់វាយតម្លៃថាតើគំរូបណ្ដុះបណ្ដាលមួយសំណុំមានគុណភាពគ្រប់គ្រាន់ដែរឬទេ មុននឹងប្រើក្នុងការចាត់ថ្នាក់។

### គ. កិច្ចការខ្លី

ក្នុងពិសោធន៍ចាត់ថ្នាក់មានការណែនាំ ចូរសាកល្បងចំនួនគំរូ ៣ ១០ និង ៣០។ សម្រាប់នីមួយៗ ចុច «យកគំរូ និងចាត់ថ្នាក់» បីដង ហើយកត់ត្រាភាពត្រឹមត្រូវទាំងបី។ គណនាមធ្យម និងគម្លាតរវាងការសាកល្បង។ សរសេរការសង្កេត៖ តើភាពប្រែប្រួល (variance) ថយចុះយ៉ាងណា ពេលចំនួនគំរូកើនឡើង?
